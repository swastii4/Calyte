import React, { useState, useEffect, useCallback, useRef } from 'react';
import { Moon, Sun, Volume2, VolumeX, Heart, Clock, Wind, Music, 
         MessageSquare, Gamepad2, Users, Flame, Bot, Shield, 
         Trophy, Calendar, User, LogIn, LogOut, Send, X } from 'lucide-react';

const CalyteLanding = () => {
  const [isBreathing, setIsBreathing] = useState(true);
  const [activeMode, setActiveMode] = useState('breathe');
  const [isAudioPlaying, setIsAudioPlaying] = useState(false);
  const [isDarkMode, setIsDarkMode] = useState(false);
  const [userStreak, setUserStreak] = useState(7);
  const [totalMinutes, setTotalMinutes] = useState(120);
  const [isLoggedIn, setIsLoggedIn] = useState(false);
  const [showProfileMenu, setShowProfileMenu] = useState(false);
  const [showChatbot, setShowChatbot] = useState(false);
  const [chatMessages, setChatMessages] = useState([
    { type: 'bot', text: 'Hello! I\'m here to support you. How are you feeling today?' }
  ]);
  const [chatInput, setChatInput] = useState('');

  const primaryRef = useRef(null);
  const secondaryRef = useRef(null);
  const rafRef = useRef(null);

  const updateGradients = useCallback(() => {
    const time = Date.now() / 6000;
    const x = 50 + Math.sin(time) * 18;
    const y = 50 + Math.cos(time * 0.9) * 18;
    const scale = 1 + Math.sin(time / 2) * 0.08;

    const primaryEl = primaryRef.current;
    const secondaryEl = secondaryRef.current;

    if (primaryEl && secondaryEl) {
      if (isDarkMode) {
        primaryEl.style.background = `radial-gradient(circle at ${x}% ${y}%,
          rgba(255, 240, 200, 0.85) 0%,
          rgba(255, 220, 120, 0.75) 8%,
          rgba(255, 190, 100, 0.65) 18%,
          rgba(90, 90, 100, 0.8) 28%,
          rgba(35, 35, 40, 0.95) 45%,
          rgba(10, 10, 12, 1) 100%
        )`;

        secondaryEl.style.background = `radial-gradient(circle at ${x}% ${y}%,
          rgba(255, 220, 120, 0.18) 0%,
          rgba(120,120,130,0.25) 18%,
          rgba(30,30,36,0.95) 50%,
          rgba(10,10,12,1) 100%
        )`;
      } else {
        primaryEl.style.background = `radial-gradient(circle at ${x}% ${y}%,
          rgba(255, 255, 255, 0.65) 0%,
          rgba(255, 245, 200, 0.75) 3%,
          rgba(255, 230, 140, 0.80) 8%,
          rgba(255, 210, 120, 0.85) 14%,
          rgba(130, 210, 255, 0.85) 45%,
          rgba(255, 180, 180, 0.85) 100%
        )`;

        secondaryEl.style.background = `radial-gradient(circle at ${x}% ${y}%,
          rgba(255,250,220,0.12) 0%,
          rgba(200,220,255,0.10) 20%,
          rgba(240,245,255,0.05) 60%,
          rgba(255,255,255,0) 100%
        )`;
      }

      primaryEl.style.transform = `scale(${scale + 0.15})`;
      secondaryEl.style.transform = `scale(${scale})`;
    }

    rafRef.current = requestAnimationFrame(updateGradients);
  }, [isDarkMode]);

  useEffect(() => {
    if (rafRef.current == null) updateGradients();
    return () => {
      if (rafRef.current) cancelAnimationFrame(rafRef.current);
      rafRef.current = null;
    };
  }, [updateGradients]);

  const features = [
    { icon: <Wind className="w-6 h-6" />, title: "Breathing Exercises", description: "Guided breathing patterns for stress relief" },
    { icon: <Music className="w-6 h-6" />, title: "Ambient Sounds", description: "Calming nature and meditation sounds" },
    { icon: <Heart className="w-6 h-6" />, title: "Mood Tracking", description: "Monitor your emotional well-being" },
    { icon: <Clock className="w-6 h-6" />, title: "Timer Sessions", description: "Customizable meditation timers" }
  ];

  const games = [
    { title: "Zen Garden", description: "Create and maintain your digital garden", duration: "5-15 min" },
    { title: "Color Flow", description: "Relaxing color matching puzzle", duration: "10-20 min" },
    { title: "Mindful Maze", description: "Navigate through peaceful labyrinths", duration: "5-10 min" },
    { title: "Pattern Peace", description: "Create calming mandala patterns", duration: "15-30 min" }
  ];

  const communityFeatures = [
    { icon: <Bot className="w-6 h-6" />, title: "AI Therapy Chat", description: "24/7 support from our AI counselor" },
    { icon: <Users className="w-6 h-6" />, title: "Support Circles", description: "Connect with others on similar journeys" },
    { icon: <Shield className="w-6 h-6" />, title: "Anonymous Chat", description: "Safe, private conversations" },
    { icon: <MessageSquare className="w-6 h-6" />, title: "Group Sessions", description: "Guided group meditation and discussions" }
  ];

  const ModeButton = ({ mode, title, icon }) => (
    <button
      onClick={() => setActiveMode(mode)}
      className={`flex items-center gap-2 px-6 py-3 rounded-lg transition-all duration-300 ${
        activeMode === mode 
          ? `${isDarkMode ? 'bg-white/20 text-white' : 'bg-slate-900/20 text-slate-900'}` 
          : `${isDarkMode ? 'bg-white/5 text-white/70 hover:bg-white/10' : 'bg-slate-900/5 text-slate-900/70 hover:bg-slate-900/10'}`
      }`}
    >
      {icon}
      {title}
    </button>
  );

  const handleSendMessage = () => {
    if (chatInput.trim()) {
      setChatMessages([...chatMessages, { type: 'user', text: chatInput }]);
      setChatInput('');
      
      setTimeout(() => {
        const responses = [
          "I understand. Take a deep breath with me. Inhale... and exhale...",
          "That sounds challenging. Would you like to try a guided meditation?",
          "Thank you for sharing. Remember, it's okay to take things one step at a time.",
          "I'm here for you. Have you tried our breathing exercises today?",
          "Your feelings are valid. Let's work through this together."
        ];
        setChatMessages(prev => [...prev, { 
          type: 'bot', 
          text: responses[Math.floor(Math.random() * responses.length)] 
        }]);
      }, 1000);
    }
  };

  return (
    <div className={`min-h-screen ${isDarkMode ? 'bg-[#07070a]' : 'bg-blue-50'} overflow-x-hidden font-sans`}>
      <div className="fixed inset-0 rounded-3xl overflow-hidden">
        <div 
          ref={primaryRef} 
          className={`absolute inset-0 rounded-3xl blur-[48px] transition-transform duration-1000 primary-gradient ${isBreathing ? 'animate-breathe' : ''}`} 
        />
        <div 
          ref={secondaryRef} 
          className={`absolute inset-0 rounded-3xl opacity-60 blur-[28px] transition-transform duration-1000 secondary-gradient ${isBreathing ? 'animate-breathe' : ''}`} 
        />

        <div className="absolute inset-0 smoke-cloud cloud-1" />
        <div className="absolute inset-0 smoke-cloud cloud-2" />
        <div className="absolute inset-0 smoke-cloud cloud-3" />

        <div className="absolute inset-0 opacity-12 mix-blend-overlay pointer-events-none noise-overlay" />
      </div>

      <nav className="relative z-10 w-full p-6">
        <div className="max-w-7xl mx-auto flex justify-between items-center">
          <h1 className={`${isDarkMode ? 'text-white/95' : 'text-slate-900'} font-serif text-3xl`}>Calyte</h1>
          <div className="flex items-center gap-6">
            <div className={`flex items-center gap-2 ${isDarkMode ? 'bg-white/5' : 'bg-slate-900/5'} px-4 py-2 rounded-lg`}>
              <Flame className="w-5 h-5 text-orange-400" />
              <span className={isDarkMode ? 'text-white/90' : 'text-slate-900/90'}>{userStreak} day streak</span>
            </div>
            <button 
              onClick={() => setIsAudioPlaying(!isAudioPlaying)} 
              className={`p-2 rounded-lg ${isDarkMode ? 'bg-white/10 text-white/90' : 'bg-slate-900/10 text-slate-900/90'}`}
            >
              {isAudioPlaying ? <Volume2 className="w-5 h-5" /> : <VolumeX className="w-5 h-5" />}
            </button>
            <button 
              onClick={() => setIsDarkMode(!isDarkMode)} 
              className={`p-2 rounded-lg ${isDarkMode ? 'bg-white/10 text-white/90' : 'bg-slate-900/10 text-slate-900/90'}`}
            >
              {isDarkMode ? <Sun className="w-5 h-5" /> : <Moon className="w-5 h-5" />}
            </button>
            <div className="relative">
              <button 
                onClick={() => setShowProfileMenu(!showProfileMenu)} 
                className={`p-2 rounded-lg ${isDarkMode ? 'bg-white/10 text-white/90' : 'bg-slate-900/10 text-slate-900/90'}`}
              >
                <User className="w-5 h-5" />
              </button>
              {showProfileMenu && (
                <div className={`absolute right-0 mt-2 w-48 ${isDarkMode ? 'bg-slate-900/95 border-white/10' : 'bg-white/95 border-slate-900/10'} backdrop-blur-md rounded-lg border shadow-lg`}>
                  {isLoggedIn ? (
                    <>
                      <div className={`px-4 py-3 border-b ${isDarkMode ? 'border-white/10 text-white/90' : 'border-slate-900/10 text-slate-900/90'}`}>
                        <p className="text-sm font-medium">Welcome back!</p>
                      </div>
                      <button
                        onClick={() => { setIsLoggedIn(false); setShowProfileMenu(false); }}
                        className={`w-full px-4 py-2 text-left flex items-center gap-2 ${isDarkMode ? 'text-white/80 hover:bg-white/10' : 'text-slate-900/80 hover:bg-slate-900/10'}`}
                      >
                        <LogOut className="w-4 h-4" />
                        Logout
                      </button>
                    </>
                  ) : (
                    <button
                      onClick={() => { setIsLoggedIn(true); setShowProfileMenu(false); }}
                      className={`w-full px-4 py-2 text-left flex items-center gap-2 ${isDarkMode ? 'text-white/80 hover:bg-white/10' : 'text-slate-900/80 hover:bg-slate-900/10'}`}
                    >
                      <LogIn className="w-4 h-4" />
                      Login
                    </button>
                  )}
                </div>
              )}
            </div>
          </div>
        </div>
      </nav>

      <div className={`relative z-10 w-full ${isDarkMode ? 'bg-white/5' : 'bg-slate-900/5'} backdrop-blur-md border-y ${isDarkMode ? 'border-white/10' : 'border-slate-900/10'}`}>
        <div className="max-w-7xl mx-auto py-4 px-6 flex justify-center gap-12">
          <div className="flex items-center gap-3">
            <Trophy className="w-5 h-5 text-yellow-400" />
            <span className={isDarkMode ? 'text-white/90' : 'text-slate-900/90'}>{totalMinutes} minutes of mindfulness</span>
          </div>
          <div className="flex items-center gap-3">
            <Calendar className="w-5 h-5 text-green-400" />
            <span className={isDarkMode ? 'text-white/90' : 'text-slate-900/90'}>Joined community of 50,000+</span>
          </div>
        </div>
      </div>

      <main className="relative z-10 max-w-7xl mx-auto px-6 pt-20 pb-32">
        <div className="text-center mb-20">
          <h2 className={`${isDarkMode ? 'text-white/95' : 'text-slate-900'} font-serif text-6xl mb-6`}>Find Your Inner Peace</h2>
          <p className={`${isDarkMode ? 'text-white/80' : 'text-slate-700'} text-xl mb-12 max-w-2xl mx-auto`}>
            Experience tranquility through meditation, games, and community support.
          </p>
          <button
            onClick={() => setIsBreathing(!isBreathing)}
            className={`${isDarkMode ? 'bg-white/10 text-white/90 hover:bg-white/20' : 'bg-slate-900/10 text-slate-900/90 hover:bg-slate-900/20'} px-8 py-4 rounded-lg text-lg tracking-wide 
              transition-all duration-300 hover:-translate-y-0.5`}
          >
            {isBreathing ? 'Pause Breathing' : 'Begin Your Journey'}
          </button>
        </div>

        <div className="flex justify-center gap-4 mb-20 flex-wrap">
          <ModeButton mode="breathe" title="Breathe" icon={<Wind className="w-5 h-5" />} />
          <ModeButton mode="meditate" title="Meditate" icon={<Heart className="w-5 h-5" />} />
          <ModeButton mode="focus" title="Focus" icon={<Clock className="w-5 h-5" />} />
          <ModeButton mode="sleep" title="Sleep" icon={<Moon className="w-5 h-5" />} />
        </div>

        <h3 className={`${isDarkMode ? 'text-white/90' : 'text-slate-900'} font-serif text-3xl mb-8`}>Core Features</h3>
        <div className="grid md:grid-cols-2 lg:grid-cols-4 gap-8 mb-20">
          {features.map((feature, index) => (
            <div key={index} className={`${isDarkMode ? 'bg-white/5 border-white/10' : 'bg-slate-900/5 border-slate-900/10'} backdrop-blur-md rounded-2xl p-6 border`}>
              <div className={isDarkMode ? 'text-white/90' : 'text-slate-900/90'}>{feature.icon}</div>
              <h3 className={`${isDarkMode ? 'text-white/90' : 'text-slate-900'} text-xl mb-2 mt-4`}>{feature.title}</h3>
              <p className={isDarkMode ? 'text-white/70' : 'text-slate-700'}>{feature.description}</p>
            </div>
          ))}
        </div>

        <h3 className={`${isDarkMode ? 'text-white/90' : 'text-slate-900'} font-serif text-3xl mb-8`}>Relaxation Games</h3>
        <div className="grid md:grid-cols-2 lg:grid-cols-4 gap-8 mb-20">
          {games.map((game, index) => (
            <div key={index} className={`${isDarkMode ? 'bg-white/5 border-white/10' : 'bg-slate-900/5 border-slate-900/10'} backdrop-blur-md rounded-2xl p-6 border`}>
              <div className={isDarkMode ? 'text-white/90' : 'text-slate-900/90'}><Gamepad2 className="w-6 h-6" /></div>
              <h3 className={`${isDarkMode ? 'text-white/90' : 'text-slate-900'} text-xl mb-2 mt-4`}>{game.title}</h3>
              <p className={`${isDarkMode ? 'text-white/70' : 'text-slate-700'} mb-3`}>{game.description}</p>
              <span className={isDarkMode ? 'text-white/50' : 'text-slate-600'}>{game.duration}</span>
            </div>
          ))}
        </div>

        <h3 className={`${isDarkMode ? 'text-white/90' : 'text-slate-900'} font-serif text-3xl mb-8`}>Community & Support</h3>
        <div className="grid md:grid-cols-2 lg:grid-cols-4 gap-8 mb-20">
          {communityFeatures.map((feature, index) => (
            <div key={index} className={`${isDarkMode ? 'bg-white/5 border-white/10' : 'bg-slate-900/5 border-slate-900/10'} backdrop-blur-md rounded-2xl p-6 border`}>
              <div className={isDarkMode ? 'text-white/90' : 'text-slate-900/90'}>{feature.icon}</div>
              <h3 className={`${isDarkMode ? 'text-white/90' : 'text-slate-900'} text-xl mb-2 mt-4`}>{feature.title}</h3>
              <p className={isDarkMode ? 'text-white/70' : 'text-slate-700'}>{feature.description}</p>
            </div>
          ))}
        </div>
      </main>

      <footer className={`relative z-10 border-t ${isDarkMode ? 'border-white/10' : 'border-slate-900/10'}`}>
        <div className="max-w-7xl mx-auto px-6 py-8">
          <div className={`${isDarkMode ? 'text-white/60' : 'text-slate-600'} text-center`}>
            © 2025 Calyte. Find your peace, anywhere, anytime.
          </div>
        </div>
      </footer>

      <button
        onClick={() => setShowChatbot(!showChatbot)}
        className={`fixed bottom-8 right-8 z-50 p-4 rounded-full shadow-2xl transition-all duration-300 hover:scale-110 ${
          isDarkMode 
            ? 'bg-gradient-to-br from-purple-500/90 to-pink-500/90 text-white' 
            : 'bg-gradient-to-br from-blue-500/90 to-purple-500/90 text-white'
        }`}
      >
        {showChatbot ? <X className="w-6 h-6" /> : <Bot className="w-6 h-6" />}
      </button>

      {showChatbot && (
        <div className={`fixed bottom-28 right-8 z-50 w-96 h-[500px] rounded-2xl shadow-2xl overflow-hidden transition-all duration-300 ${
          isDarkMode ? 'bg-slate-900/95 border-white/10' : 'bg-white/95 border-slate-900/10'
        } backdrop-blur-xl border`}>
          <div className={`p-4 border-b ${isDarkMode ? 'border-white/10 bg-gradient-to-r from-purple-500/20 to-pink-500/20' : 'border-slate-900/10 bg-gradient-to-r from-blue-500/20 to-purple-500/20'}`}>
            <div className="flex items-center gap-3">
              <div className={`p-2 rounded-full ${isDarkMode ? 'bg-purple-500/30' : 'bg-blue-500/30'}`}>
                <Bot className="w-5 h-5" />
              </div>
              <div>
                <h3 className={`font-medium ${isDarkMode ? 'text-white/90' : 'text-slate-900'}`}>AI Therapy Assistant</h3>
                <p className={`text-xs ${isDarkMode ? 'text-white/60' : 'text-slate-600'}`}>Always here to listen</p>
              </div>
            </div>
          </div>

          <div className="h-[340px] overflow-y-auto p-4 space-y-4">
            {chatMessages.map((msg, idx) => (
              <div key={idx} className={`flex ${msg.type === 'user' ? 'justify-end' : 'justify-start'}`}>
                <div className={`max-w-[80%] p-3 rounded-2xl ${
                  msg.type === 'user'
                    ? isDarkMode 
                      ? 'bg-gradient-to-r from-purple-500/30 to-pink-500/30 text-white/90' 
                      : 'bg-gradient-to-r from-blue-500/30 to-purple-500/30 text-slate-900'
                    : isDarkMode
                      ? 'bg-white/5 text-white/80'
                      : 'bg-slate-900/5 text-slate-800'
                }`}>
                  {msg.text}
                </div>
              </div>
            ))}
          </div>

          <div className={`p-4 border-t ${isDarkMode ? 'border-white/10' : 'border-slate-900/10'}`}>
            <div className="flex gap-2">
              <input
                type="text"
                value={chatInput}
                onChange={(e) => setChatInput(e.target.value)}
                onKeyPress={(e) => e.key === 'Enter' && handleSendMessage()}
                placeholder="Share your thoughts..."
                className={`flex-1 px-4 py-2 rounded-lg focus:outline-none focus:ring-2 transition-all ${
                  isDarkMode 
                    ? 'bg-white/5 text-white/90 placeholder-white/40 focus:ring-purple-500/50' 
                    : 'bg-slate-900/5 text-slate-900 placeholder-slate-500 focus:ring-blue-500/50'
                }`}
              />
              <button
                onClick={handleSendMessage}
                className={`p-2 rounded-lg transition-all hover:scale-105 ${
                  isDarkMode 
                    ? 'bg-gradient-to-r from-purple-500/30 to-pink-500/30 text-white/90 hover:from-purple-500/40 hover:to-pink-500/40' 
                    : 'bg-gradient-to-r from-blue-500/30 to-purple-500/30 text-slate-900 hover:from-blue-500/40 hover:to-purple-500/40'
                }`}
              >
                <Send className="w-5 h-5" />
              </button>
            </div>
          </div>
        </div>
      )}

      <style jsx global>{`
        @import url('https://fonts.googleapis.com/css2?family=Cormorant+Garamond:wght@300;400;500&family=Montserrat:wght@300;400;500&display=swap');

        body {
          font-family: 'Montserrat', system-ui, -apple-system, sans-serif;
          margin: 0;
        }

        h1, h2, h3 {
          font-family: 'Cormorant Garamond', serif;
        }

        @keyframes breathe {
          0%, 100% { transform: scale(1); }
          50% { transform: scale(1.04); }
        }

        .animate-breathe {
          animation: breathe 4.2s ease-in-out infinite;
        }

        @keyframes float-cloud-1 {
          0%, 100% { transform: translate(0, 0) scale(1); opacity: 0.02; }
          25% { transform: translate(20px, -12px) scale(1.05); opacity: 0.04; }
          50% { transform: translate(40px, 8px) scale(1.03); opacity: 0.03; }
          75% { transform: translate(10px, 18px) scale(0.98); opacity: 0.05; }
        }

        @keyframes float-cloud-2 {
          0%,100% { transform: translate(0,0) scale(1) rotate(0deg); opacity: 0.03; }
          33% { transform: translate(-30px, 18px) scale(1.08) rotate(4deg); opacity: 0.045; }
          66% { transform: translate(-12px, -22px) scale(0.92) rotate(-2deg); opacity: 0.035; }
        }

        @keyframes float-cloud-3 {
          0%,100% { transform: translate(0,0) scale(1); opacity: 0.03; }
          20% { transform: translate(30px, 30px) scale(1.12); opacity: 0.02; }
          40% { transform: translate(-22px, 16px) scale(0.9); opacity: 0.045; }
          60% { transform: translate(18px, -18px) scale(1.05); opacity: 0.03; }
          80% { transform: translate(-12px, -32px) scale(0.97); opacity: 0.04; }
        }

        .smoke-cloud { pointer-events: none; mix-blend-mode: overlay; filter: blur(64px); }

        .cloud-1 {
          background: radial-gradient(ellipse 800px 600px at 20% 30%, rgba(255, 255, 255, 0.12) 0%, rgba(200, 200, 255, 0.08) 30%, transparent 70%);
          animation: float-cloud-1 25s ease-in-out infinite;
        }

        .cloud-2 {
          background: radial-gradient(ellipse 700px 700px at 80% 60%, rgba(255, 255, 255, 0.10) 0%, rgba(255, 200, 255, 0.06) 25%, transparent 65%);
          animation: float-cloud-2 30s ease-in-out infinite;
          animation-delay: -10s;
        }

        .cloud-3 {
          background: radial-gradient(ellipse 900px 500px at 50% 80%, rgba(255, 255, 255, 0.09) 0%, rgba(200, 255, 255, 0.05) 35%, transparent 70%);
          animation: float-cloud-3 35s ease-in-out infinite;
          animation-delay: -20s;
        }

        .noise-overlay {
          background-image: url("data:image/svg+xml,%3Csvg viewBox='0 0 256 256' xmlns='http://www.w3.org/2000/svg'%3E%3Cfilter id='noise' x='0' y='0'%3E%3CfeTurbulence type='fractalNoise' baseFrequency='0.65' numOctaves='3' stitchTiles='stitch'/%3E%3C/filter%3E%3Crect width='100%25' height='100%25' filter='url(%23noise)'/%3E%3C/svg%3E");
          background-repeat: repeat;
        }
      `}</style>
    </div>
  );
};

export default CalyteLanding;
